1 - TCP (Transmission Control Protocol)

TCP is connection-oriented and ensures data is received in order and without errors.

  • Useful when data integrity is critical.
  • Examples: HTTP/HTTPS, SMTP, IMAP, POP3, FTP, SSH.
TCP provides reliable delivery, but it is generally slower than UDP due to connection and acknowledgment overhead.
⚡

2 - UDP (User Datagram Protocol)

UDP is connectionless and faster, but less reliable than TCP.

  • Does not guarantee delivery, order, or error-free transmission.
  • Great for real-time traffic like streaming, online gaming, and VoIP.
Use UDP when speed is more important than guaranteed delivery.
🖥️

3 - TCP Socket Example

Python TCP Server
import socket

s = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
s.bind(('localhost', 12345))
s.listen(1)

print("Server listening on port 12345")

conn, addr = s.accept()
print('Got connection from', addr)

conn.sendall(b'Hello, client!')

data = conn.recv(1024)
print('Received', repr(data))

conn.close()
s.close()

4 - UDP Socket Example

Python UDP Server
import socket

s = socket.socket(socket.AF_INET, socket.SOCK_DGRAM)
s.bind(('localhost', 12346))

print("UDP server listening on port 12346")

while True:
		data, addr = s.recvfrom(1024)
		print('Received', repr(data), 'from', addr)
		s.sendto(b'Hello, client!', addr)

5 - Tic Tac Toe Game Example (TCP)

A simple Tic Tac Toe game can be built using TCP sockets in Python.

Server Code

Python Server
import socket
import threading

board = [' ' for _ in range(9)]
players = ['X', 'O']
current_player = 0

def handle_client(client_socket, player):
		global current_player
		print(f"Player {player} connected.")
		while True:
				try:
						move = client_socket.recv(1024).decode()
						if move in '123456789':
								if board[int(move) - 1] == ' ':
										board[int(move) - 1] = players[player]
										client_socket.sendall(str(board).encode())
										if check_winner(players[player]):
												client_socket.sendall("You win!".encode())
												break
										current_player = 1 - current_player
								else:
										client_socket.sendall("Position already taken!".encode())
						else:
								client_socket.sendall("Invalid move!".encode())
				except ConnectionResetError:
						break

		print(f"Player {player} disconnected.")
		client_socket.close()

def check_winner(player):
		winning_combinations = [
				(0, 1, 2), (3, 4, 5), (6, 7, 8),
				(0, 3, 6), (1, 4, 7), (2, 5, 8),
				(0, 4, 8), (2, 4, 6)
		]
		for combo in winning_combinations:
				if board[combo[0]] == board[combo[1]] == board[combo[2]] == player:
						return True
		return False

def start_server():
		global board
		board = [' ' for _ in range(9)]
		s = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
		s.bind(('localhost', 12345))
		s.listen(2)
		print("Server is listening on port 12345")

		for i in range(2):
				client_socket, addr = s.accept()
				threading.Thread(target=handle_client, args=(client_socket, i)).start()

		s.close()

if __name__ == "__main__":
		start_server()

Client Code

Python Client
import socket

def main():
		client_socket = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
		client_socket.connect(('localhost', 12345))
		player = client_socket.recv(1024).decode()
		print(f"You are Player {player}")

		while True:
				try:
						board_str = client_socket.recv(1024).decode()
						print(board_str)

						move = input("Enter your move (1-9): ")
						client_socket.sendall(move.encode())

						response = client_socket.recv(1024).decode()
						print(response)
						if "win" in response:
								break
				except ConnectionResetError:
						break

		client_socket.close()

if __name__ == "__main__":
		main()
